
#include <stdio.h>
#include <stdlib.h>
#include <pthread.h>

#define N 4000000
#define T 4

int sales[N];
long sales_total = 0;

pthread_mutex_t lock = PTHREAD_MUTEX_INITIALIZER;

typedef struct {
    int start;
    int end;
} range_t;

void *worker(void *arg) {
    range_t *r = (range_t *)arg;
    long local = 0;

    for (int i = r->start; i < r->end; i++) {
        local += sales[i];
    }

    pthread_mutex_lock(&lock);
    sales_total += local;
    pthread_mutex_unlock(&lock);

    return NULL;
}

int main(void) {
    pthread_t th[T];
    range_t rg[T];
    long expected = 0;

    srand(1);

    for (int i = 0; i < N; i++) {
        sales[i] = rand() % 100;
        expected += sales[i];
    }

    for (int t = 0; t < T; t++) {
        rg[t].start = t * (N / T);
        rg[t].end = (t + 1) * (N / T);

        pthread_create(&th[t], NULL, worker, &rg[t]);
    }

    for (int t = 0; t < T; t++) {
        pthread_join(th[t], NULL);
    }

    printf("Expected: %ld\n", expected);
    printf("Got:      %ld\n", sales_total);

    pthread_mutex_destroy(&lock);

    return 0;
}
